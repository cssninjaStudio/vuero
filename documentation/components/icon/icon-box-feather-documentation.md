### Feather Icons

The Vuero `<IconBox />` component works also very well with the `Feather Icons` icon set. Use the same markup and just replace the existing icons with some from Feather Icons.

<!--code-->

```vue {2-4}
<template>
  <IconBox size="medium" color="primary" rounded>
    <i class="iconify" data-icon="feather:award"></i>
  </IconBox>
  <IconBox size="medium" color="info" rounded>
    <i class="iconify" data-icon="feather:chrome"></i>
  </IconBox>
  <IconBox size="medium" color="success" rounded>
    <i class="iconify" data-icon="feather:clock"></i>
  </IconBox>
  <IconBox size="medium" color="warning" rounded>
    <i class="iconify" data-icon="feather:coffee"></i>
  </IconBox>
  <IconBox size="medium" color="danger" rounded>
    <i class="iconify" data-icon="feather:home"></i>
  </IconBox>
  <IconBox size="medium" color="purple" rounded>
    <i class="iconify" data-icon="feather:lock"></i>
  </IconBox>
  <IconBox size="medium" color="yellow" rounded>
    <i class="iconify" data-icon="feather:moon"></i>
  </IconBox>
  <IconBox size="medium" color="orange" rounded>
    <i class="iconify" data-icon="feather:phone"></i>
  </IconBox>
  <IconBox size="medium" color="green" rounded>
    <i class="iconify" data-icon="feather:paperclip"></i>
  </IconBox>
  <IconBox size="medium" color="red" rounded>
    <i class="iconify" data-icon="feather:smile"></i>
  </IconBox>
  <IconBox size="medium" color="blue" rounded>
    <i class="iconify" data-icon="feather:shield"></i>
  </IconBox>
</template>
```
